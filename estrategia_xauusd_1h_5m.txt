indicator("Estrategia XAUUSD · Paso 1 (1H) + Paso 2 (5m)", shorttitle = "Paso 1+2", overlay = true, max_boxes_count = 500, max_lines_count = 500, max_labels_count = 500)

// =====================================================================
// AJUSTES
// =====================================================================
grpN     = "Altos y bajos / Manipulaciones"
pivLen   = input.int(3, "Velas a cada lado para confirmar un alto/bajo", minval = 1, group = grpN)
maxVelas = input.int(4, "Velas máximas para volver dentro (manipulación)", minval = 1, group = grpN)
colNivel = input.color(color.gray, "Color altos/bajos sin tocar", group = grpN)
colManip = input.color(color.red, "Color manipulación", group = grpN)

grpF     = "Vacíos (FVG)"
showFvg  = input.bool(true, "Mostrar vacíos", group = grpF)
colFvgUp = input.color(color.new(#42A5F5, 80), "Vacío alcista", group = grpF)
colFvgDn = input.color(color.new(#1565C0, 80), "Vacío bajista", group = grpF)

grpO     = "Order Blocks"
showOb   = input.bool(true, "Mostrar order blocks", group = grpO)
colObUp  = input.color(color.new(#FF9800, 80), "OB alcista", group = grpO)
colObDn  = input.color(color.new(#E65100, 80), "OB bajista", group = grpO)

grpG     = "General"
maxDentro = input.int(4, "OB: máximo de velas seguidas dentro (si hay más, se invalida)", minval = 1, group = grpO)
maxZonas = input.int(15, "Máximo de vacíos / OB activos (de cada tipo)", minval = 1, group = grpG)
maxNivel = input.int(30, "Máximo de altos/bajos activos", minval = 2, group = grpG)

// =====================================================================
// TIPOS
// =====================================================================
// Un alto o un bajo de estructura que todavía no ha sido manipulado
type Nivel
    line  ln
    float precio
    bool  esAlto
    int   x1
    int   estado = 0      // 0 = esperando, 1 = una vela cerró fuera (contando velas)
    int   cuenta = 0      // velas pasadas desde que cerró fuera
    float extremo = na    // punta máxima (alto) o mínima (bajo) a la que llegó la manipulación
    bool  vivo   = true

// Un vacío (FVG) o un order block
type Zona
    box   bx
    line  mid
    float top
    float bot
    bool  alcista
    bool  esFvg
    bool  tocado = false  // la vela anterior tocó la zona (50 % en FVG, cuerpo en OB)
    int   dentro = 0      // velas seguidas dentro de la zona (solo se usa en OB)
    color col    = na     // color de la zona (para poder ocultarla y volver a mostrarla)

// =====================================================================
// MOTOR DE LA DIRECCIÓN DE 1H (sin dibujos)
// Es una copia exacta de las reglas del paso 1, pero sin dibujar nada,
// para poder calcular la dirección de 1H desde 5 minutos (o cualquier temporalidad menor).
// =====================================================================
type NivelP
    float precio
    bool  esAlto
    int   estado  = 0
    int   cuenta  = 0
    float extremo = na
    bool  vivo    = true

type ZonaP
    float top
    float bot
    bool  alcista
    bool  esFvg
    bool  tocado = false
    int   dentro = 0

procZonasP(array<ZonaP> arr) =>
    int   dir   = 0
    float borde = na
    if arr.size() > 0
        for i = arr.size() - 1 to 0
            z    = arr.get(i)
            mid  = (z.top + z.bot) / 2
            toca = z.esFvg ? (z.alcista ? low <= mid : high >= mid) : (z.alcista ? low <= z.top : high >= z.bot)
            invalida = z.alcista ? close < z.bot : close > z.top
            enZona   = low <= z.top and high >= z.bot
            z.dentro := enZona ? z.dentro + 1 : 0
            atascado = not z.esFvg and z.dentro > maxDentro
            aFavor    = z.alcista ? close > open : close < open
            reacciona = aFavor and (toca or z.tocado)
            borrar = false
            if invalida or atascado
                borrar := true
            else if reacciona
                dir    := z.alcista ? 1 : -1
                borde  := z.alcista ? z.bot : z.top
                borrar := true
            else
                z.tocado := toca
            if borrar
                arr.remove(i)
    [dir, borde]

objetivoP(int d, array<NivelP> nv, array<ZonaP> ob, array<ZonaP> fv) =>
    float mejorDist = na
    float toque     = na
    float umbral    = na
    for n in nv
        if n.vivo and (d == 1 ? (n.esAlto and n.precio > close) : (not n.esAlto and n.precio < close))
            dist = math.abs(n.precio - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                toque     := n.precio
                umbral    := n.precio
    for z in ob
        if d == 1 ? (not z.alcista and z.bot > close) : (z.alcista and z.top < close)
            dist = math.abs((d == 1 ? z.bot : z.top) - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                toque     := d == 1 ? z.bot : z.top
                umbral    := d == 1 ? z.top : z.bot
    for z in fv
        if d == 1 ? (not z.alcista and z.bot > close) : (z.alcista and z.top < close)
            dist = math.abs((d == 1 ? z.bot : z.top) - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                toque     := d == 1 ? z.bot : z.top
                umbral    := d == 1 ? z.top : z.bot
    [toque, umbral]

// Devuelve la dirección: 1 = compra, -1 = venta, 0 = sin dirección
motorDir() =>
    var array<NivelP> nv = array.new<NivelP>()
    var array<ZonaP>  fv = array.new<ZonaP>()
    var array<ZonaP>  ob = array.new<ZonaP>()
    var int    dir     = 0
    var float  oOrigen = na
    var float  oToque  = na
    var float  oUmbral = na
    var int    ultTipo = 0
    var NivelP uA      = na
    var NivelP uB      = na
    var int    rIdx    = na
    var float  rO      = na
    var float  rC      = na
    var int    vIdx    = na
    var float  vO      = na
    var float  vC      = na
    nueva = false
    // Cancelaciones
    if dir != 0 and not na(oOrigen) and (dir == 1 ? close < oOrigen : close > oOrigen)
        dir     := 0
        oOrigen := na
        oToque  := na
        oUmbral := na
    if dir != 0 and not na(oUmbral)
        if (dir == 1 ? close > oUmbral : close < oUmbral) or (dir == 1 ? high >= oToque : low <= oToque)
            dir     := 0
            oOrigen := na
            oToque  := na
            oUmbral := na
    libre = dir == 0 or na(oUmbral)
    // Manipulaciones
    if nv.size() > 0
        for i = nv.size() - 1 to 0
            n = nv.get(i)
            if not n.vivo
                nv.remove(i)
                continue
            manip = false
            roto  = false
            cf    = false
            dentro = n.esAlto ? close <= n.precio : close >= n.precio
            if n.estado == 0
                pasa = n.esAlto ? high > n.precio : low < n.precio
                if pasa
                    n.extremo := n.esAlto ? high : low
                manip := pasa and dentro
                cf    := pasa and not dentro
            else
                n.cuenta  := n.cuenta + 1
                n.extremo := n.esAlto ? math.max(n.extremo, high) : math.min(n.extremo, low)
                manip := dentro
                roto  := not dentro and n.cuenta >= maxVelas
            if cf
                n.estado := 1
            if manip or roto
                dM = n.esAlto ? -1 : 1
                if manip and (libre or dM == -dir)
                    dir     := dM
                    oOrigen := n.extremo
                    nueva   := true
                    libre   := false
                n.vivo := false
                nv.remove(i)
    // Altos y bajos alternados
    ph = ta.pivothigh(high, pivLen, pivLen)
    pl = ta.pivotlow(low, pivLen, pivLen)
    if not na(ph)
        rep = ultTipo == 1
        if not rep or ph > uA.precio
            if rep and uA.vivo and uA.estado == 0
                uA.vivo := false
            uA := NivelP.new(ph, true)
            nv.push(uA)
            ultTipo := 1
    if not na(pl)
        rep = ultTipo == -1
        if not rep or pl < uB.precio
            if rep and uB.vivo and uB.estado == 0
                uB.vivo := false
            uB := NivelP.new(pl, false)
            nv.push(uB)
            ultTipo := -1
    while nv.size() > maxNivel
        v = nv.shift()
        v.vivo := false
    // Reacciones en vacíos y OB
    [dF, bF] = procZonasP(fv)
    if dF != 0 and (libre or dF == -dir)
        dir     := dF
        oOrigen := bF
        nueva   := true
        libre   := false
    [dO, bO] = procZonasP(ob)
    if dO != 0 and (libre or dO == -dir)
        dir     := dO
        oOrigen := bO
        nueva   := true
        libre   := false
    // Vacíos nuevos
    if showFvg
        if low > high[2]
            fv.push(ZonaP.new(low, high[2], true, true))
        if high < low[2]
            fv.push(ZonaP.new(low[2], high, false, true))
    while fv.size() > maxZonas
        fv.shift()
    // OB nuevos
    if not na(rIdx) and close > rO
        if showOb
            ob.push(ZonaP.new(rO, rC, true, false))
        rIdx := na
    if not na(vIdx) and close < vO
        if showOb
            ob.push(ZonaP.new(vC, vO, false, false))
        vIdx := na
    while ob.size() > maxZonas
        ob.shift()
    if close < open
        rIdx := bar_index
        rO   := open
        rC   := close
    if close > open
        vIdx := bar_index
        vO   := open
        vC   := close
    // Objetivo de la dirección nueva
    if nueva
        [tq, u] = objetivoP(dir, nv, ob, fv)
        oToque  := tq
        oUmbral := u
        if not na(tq) and (dir == 1 ? high >= tq : low <= tq)
            dir     := 0
            oOrigen := na
            oToque  := na
            oUmbral := na
    dir

// =====================================================================
// TEMPORALIDAD
//   - 1H: paso 1 (se marca todo + panel de dirección)
//   - Menos de 1H (5m, etc.): paso 2 (solo lo que va a favor de la dirección de 1H)
//   - Más de 1H: no se marca nada
// =====================================================================
esH1    = timeframe.period == "60"
esMenor = timeframe.isintraday and timeframe.in_seconds() < 3600
activo  = esH1 or esMenor

// Dirección de la última vela CERRADA de 1H (no cambia a mitad de vela)
dirHTF = nz(request.security(syminfo.tickerid, "60", motorDir()[1], lookahead = barmerge.lookahead_on))

// ¿Se debe ver una zona / un nivel?  En 1H todo; en temporalidades menores solo lo que va a favor de 1H.
zonaVisible(bool alcista) =>
    esH1 or (dirHTF != 0 and dirHTF == (alcista ? 1 : -1))
// En compra interesan los bajos (manipulación de bajo); en venta, los altos.
nivelVisible(bool esAlto) =>
    esH1 or (dirHTF != 0 and dirHTF == (esAlto ? -1 : 1))

transp = color.new(color.white, 100)

var array<Nivel> niveles = array.new<Nivel>()
var array<Zona>  fvgs    = array.new<Zona>()
var array<Zona>  obs     = array.new<Zona>()

// =====================================================================
// FUNCIONES
// =====================================================================
nuevoNivel(float p, bool esAlto, int x) =>
    ln = line.new(x, p, bar_index, p, color = nivelVisible(esAlto) ? colNivel : transp, style = line.style_dotted, extend = extend.right)
    Nivel.new(ln, p, esAlto, x)

// Muestra u oculta una zona (oculta = colores transparentes)
mostrarZona(Zona z, bool vis) =>
    z.bx.set_bgcolor(vis ? z.col : color.new(z.col, 100))
    z.bx.set_border_color(color.new(z.col, vis ? 40 : 100))
    z.bx.set_text_color(color.new(z.col, vis ? 0 : 100))
    if not na(z.mid)
        z.mid.set_color(color.new(z.col, vis ? 0 : 100))

nuevaZona(float top, float bot, bool alc, bool fvg, int x) =>
    c  = fvg ? (alc ? colFvgUp : colFvgDn) : (alc ? colObUp : colObDn)
    bx = box.new(x, top, bar_index, bot, border_color = color.new(c, 40), bgcolor = c, extend = extend.right, text = fvg ? "FVG" : "OB", text_size = size.tiny, text_color = color.new(c, 0), text_halign = text.align_left)
    line m = na
    if fvg
        // Línea del 50 % del vacío
        m := line.new(x, (top + bot) / 2, bar_index, (top + bot) / 2, color = color.new(c, 0), style = line.style_dashed, extend = extend.right)
    z = Zona.new(bx, m, top, bot, alc, fvg, col = c)
    if not zonaVisible(alc)
        mostrarZona(z, false)
    z

borrarZona(Zona z) =>
    z.bx.delete()
    if not na(z.mid)
        z.mid.delete()

// Revisa cada zona activa: invalidación y reacción.
// Devuelve [dirección, borde de origen]: 1 = compra, -1 = venta, 0 = ninguna.
// El borde de origen es el lado de la zona que, si se cierra más allá, cancela la dirección.
procesarZonas(array<Zona> arr) =>
    int   dir   = 0
    float borde = na
    if arr.size() > 0
        for i = arr.size() - 1 to 0
            z   = arr.get(i)
            mid = (z.top + z.bot) / 2
            // FVG: el precio tiene que llegar al 50 % o más. OB: basta con tocar el cuerpo.
            toca = z.esFvg ? (z.alcista ? low <= mid : high >= mid) : (z.alcista ? low <= z.top : high >= z.bot)
            // Invalidación: una vela cierra al otro lado de la zona
            invalida = z.alcista ? close < z.bot : close > z.top
            // OB: si hay más de "maxDentro" velas seguidas dentro del cuerpo, no lo va a respetar: se invalida
            enZona   = low <= z.top and high >= z.bot
            z.dentro := enZona ? z.dentro + 1 : 0
            atascado = not z.esFvg and z.dentro > maxDentro
            // Reacción: la vela que toca cierra a favor, o la vela anterior tocó y esta cierra a favor
            aFavor    = z.alcista ? close > open : close < open
            reacciona = aFavor and (toca or z.tocado)
            borrar = false
            if invalida or atascado
                borrar := true
            else if reacciona
                // Solo vale una reacción: se marca con ✓ y se borra la zona
                c = z.esFvg ? (z.alcista ? colFvgUp : colFvgDn) : (z.alcista ? colObUp : colObDn)
                if zonaVisible(z.alcista)
                    label.new(bar_index, z.alcista ? low : high, "✓ " + (z.esFvg ? "FVG" : "OB"), style = z.alcista ? label.style_label_up : label.style_label_down, color = color.new(c, 0), textcolor = color.white, size = size.tiny)
                dir    := z.alcista ? 1 : -1
                borde  := z.alcista ? z.bot : z.top
                borrar := true
            else
                z.tocado := toca
            if borrar
                borrarZona(z)
                arr.remove(i)
    [dir, borde]

limitar(array<Zona> arr) =>
    while arr.size() > maxZonas
        borrarZona(arr.shift())

// =====================================================================
// DIRECCIÓN DE 1H
// Mientras una dirección está activa, se ignoran las señales a favor.
// Una señal CONTRARIA (manipulación, vacío u OB) anula la dirección y pone la nueva.
// La dirección termina cuando el precio llega al objetivo o una vela cierra más allá.
// =====================================================================
var int    dir1h    = 0     // 1 = compra, -1 = venta, 0 = todavía sin señal
var string origen   = ""
var int    tSenal   = na
var float  objOrigen = na   // si una vela de 1H cierra más allá del origen de la señal (en contra), se cancela
var float  objToque  = na   // precio donde el precio LLEGA al objetivo
var float  objUmbral = na   // si una vela de 1H CIERRA más allá de este precio, la dirección se cancela
var string objTxt    = ""
nuevaSenal = false

// Precio en contra: cierra más allá del punto donde nació la señal
if dir1h != 0 and not na(objOrigen) and (dir1h == 1 ? close < objOrigen : close > objOrigen)
    dir1h     := 0
    origen    := "Cancelada: cerró más allá del origen de la señal"
    tSenal    := time
    objOrigen := na
    objToque  := na
    objUmbral := na
    objTxt    := ""

if dir1h != 0 and not na(objUmbral)
    cierraMas = dir1h == 1 ? close > objUmbral : close < objUmbral
    llega     = dir1h == 1 ? high >= objToque : low <= objToque
    if cierraMas or llega
        origen    := cierraMas ? "Cancelada: cerró más allá del objetivo" : "Objetivo alcanzado"
        dir1h     := 0
        tSenal    := time
        objOrigen := na
        objToque  := na
        objUmbral := na
        objTxt    := ""

// Solo se acepta una señal nueva si no hay dirección activa (o si la activa no tiene objetivo)
libre = dir1h == 0 or na(objUmbral)

// =====================================================================
// 1) MANIPULACIONES SOBRE LOS ALTOS Y BAJOS ACTIVOS
// =====================================================================
if niveles.size() > 0
    for i = niveles.size() - 1 to 0
        n = niveles.get(i)
        if not n.vivo
            niveles.remove(i)
            continue
        manip      = false
        roto       = false
        cerroFuera = false
        // ¿La vela cierra dentro del nivel (por debajo del alto / por encima del bajo)?
        dentro = n.esAlto ? close <= n.precio : close >= n.precio
        if n.estado == 0
            // Pasa el nivel: si cierra dentro es manipulación con mecha; si cierra fuera, empieza la cuenta
            pasa = n.esAlto ? high > n.precio : low < n.precio
            if pasa
                n.extremo := n.esAlto ? high : low
            manip      := pasa and dentro
            cerroFuera := pasa and not dentro
        else
            // Cerró fuera: tiene como máximo "maxVelas" velas para volver a cerrar dentro
            n.cuenta  := n.cuenta + 1
            n.extremo := n.esAlto ? math.max(n.extremo, high) : math.min(n.extremo, low)
            manip    := dentro
            roto     := not dentro and n.cuenta >= maxVelas
        if cerroFuera
            n.estado := 1
        if manip
            if nivelVisible(n.esAlto)
                label.new(bar_index, n.precio, "M", style = n.esAlto ? label.style_label_down : label.style_label_up, color = colManip, textcolor = color.white, size = size.small)
                n.ln.set_extend(extend.none)
                n.ln.set_x2(bar_index)
                n.ln.set_color(colManip)
                n.ln.set_style(line.style_solid)
                n.ln.set_width(2)
            else
                n.ln.delete()
            // Manipulación de un alto = venta. Manipulación de un bajo = compra.
            // Una manipulación contraria anula la dirección actual y pone la nueva.
            dirM = n.esAlto ? -1 : 1
            if libre or dirM == -dir1h
                dir1h      := n.esAlto ? -1 : 1
                origen     := n.esAlto ? "Manipulación de alto" : "Manipulación de bajo"
                tSenal     := time
                objOrigen  := n.extremo
                nuevaSenal := true
                libre      := false
            n.vivo := false
            niveles.remove(i)
        else if roto
            // Rompimiento real: no se marca, solo se quita el nivel
            n.ln.delete()
            n.vivo := false
            niveles.remove(i)

// =====================================================================
// 2) NUEVOS ALTOS Y BAJOS (siempre alternados: alto → bajo → alto → bajo)
// =====================================================================
ph = ta.pivothigh(high, pivLen, pivLen)
pl = ta.pivotlow(low, pivLen, pivLen)

var int   ultimoTipo = 0     // 1 = el último fue un alto, -1 = el último fue un bajo
var Nivel ultAlto    = na
var Nivel ultBajo    = na

if activo and not na(ph)
    // Dos altos seguidos sin bajo en medio: se queda solo el más alto
    repetido = ultimoTipo == 1
    agregar  = not repetido or ph > ultAlto.precio
    if agregar
        if repetido and ultAlto.vivo and ultAlto.estado == 0
            ultAlto.ln.delete()
            ultAlto.vivo := false
        ultAlto := nuevoNivel(ph, true, bar_index - pivLen)
        niveles.push(ultAlto)
        ultimoTipo := 1

if activo and not na(pl)
    // Dos bajos seguidos sin alto en medio: se queda solo el más bajo
    repetido = ultimoTipo == -1
    agregar  = not repetido or pl < ultBajo.precio
    if agregar
        if repetido and ultBajo.vivo and ultBajo.estado == 0
            ultBajo.ln.delete()
            ultBajo.vivo := false
        ultBajo := nuevoNivel(pl, false, bar_index - pivLen)
        niveles.push(ultBajo)
        ultimoTipo := -1

// Limita la cantidad de altos/bajos en pantalla
while niveles.size() > maxNivel
    viejo = niveles.shift()
    if viejo.vivo
        viejo.ln.delete()
        viejo.vivo := false

// =====================================================================
// PASO 2: cuando cambia la dirección de 1H, se muestran solo las zonas y niveles a favor
// =====================================================================
if esMenor and dirHTF != dirHTF[1]
    for z in fvgs
        mostrarZona(z, zonaVisible(z.alcista))
    for z in obs
        mostrarZona(z, zonaVisible(z.alcista))
    for n in niveles
        n.ln.set_color(nivelVisible(n.esAlto) ? colNivel : transp)

// =====================================================================
// 3) VACÍOS Y ORDER BLOCKS: primero se revisan los que ya existen...
// =====================================================================
[dirFvg, bordeFvg] = procesarZonas(fvgs)
// Una señal CONTRARIA (vacío u OB) anula la dirección actual: empieza la distribución hacia el otro lado
if dirFvg != 0 and (libre or dirFvg == -dir1h)
    dir1h      := dirFvg
    origen     := dirFvg == 1 ? "Reacción en vacío alcista" : "Reacción en vacío bajista"
    tSenal     := time
    objOrigen  := bordeFvg
    nuevaSenal := true
    libre      := false
[dirOb, bordeOb] = procesarZonas(obs)
if dirOb != 0 and (libre or dirOb == -dir1h)
    dir1h      := dirOb
    origen     := dirOb == 1 ? "Reacción en OB alcista" : "Reacción en OB bajista"
    tSenal     := time
    objOrigen  := bordeOb
    nuevaSenal := true
    libre      := false

// ...y luego se buscan los nuevos (así empiezan a contar desde la vela siguiente)

// Vacíos de 3 velas
if showFvg and activo
    if low > high[2]
        fvgs.push(nuevaZona(low, high[2], true, true, bar_index - 2))
    if high < low[2]
        fvgs.push(nuevaZona(low[2], high, false, true, bar_index - 2))
    limitar(fvgs)

// Order blocks: última vela contraria antes del movimiento
var int   rojaIdx  = na
var float rojaO    = na
var float rojaC    = na
var int   verdeIdx = na
var float verdeO   = na
var float verdeC   = na

// OB alcista: una vela cierra por encima del CUERPO de la última vela roja (su apertura)
if not na(rojaIdx) and close > rojaO
    if showOb and activo
        obs.push(nuevaZona(rojaO, rojaC, true, false, rojaIdx))
    rojaIdx := na

// OB bajista: una vela cierra por debajo del CUERPO de la última vela verde (su apertura)
if not na(verdeIdx) and close < verdeO
    if showOb and activo
        obs.push(nuevaZona(verdeC, verdeO, false, false, verdeIdx))
    verdeIdx := na

limitar(obs)

// Guarda la última vela roja y la última vela verde
if close < open
    rojaIdx := bar_index
    rojaO   := open
    rojaC   := close
if close > open
    verdeIdx := bar_index
    verdeO   := open
    verdeC   := close

// =====================================================================
// OBJETIVO DE LA DIRECCIÓN: el más cercano entre
//   - compra: el siguiente alto, OB bajista o vacío bajista por encima
//   - venta:  el siguiente bajo, OB alcista o vacío alcista por debajo
// Devuelve [precio donde se llega, precio de cancelación, texto]
// =====================================================================
buscarObjetivo(int d) =>
    float mejorDist = na
    float toque     = na
    float umbral    = na
    string txt      = "Sin objetivo a la vista"
    for n in niveles
        if n.vivo and (d == 1 ? (n.esAlto and n.precio > close) : (not n.esAlto and n.precio < close))
            dist = math.abs(n.precio - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                toque     := n.precio
                umbral    := n.precio
                txt       := (d == 1 ? "Alto " : "Bajo ") + str.tostring(n.precio, format.mintick)
    for z in obs
        if d == 1 ? (not z.alcista and z.bot > close) : (z.alcista and z.top < close)
            dist = math.abs((d == 1 ? z.bot : z.top) - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                // Llegar al OB = tocar su cuerpo. Pasarlo = cerrar más allá de todo el cuerpo.
                toque     := d == 1 ? z.bot : z.top
                umbral    := d == 1 ? z.top : z.bot
                txt       := (d == 1 ? "OB bajista " : "OB alcista ") + str.tostring(z.bot, format.mintick) + " – " + str.tostring(z.top, format.mintick)
    for z in fvgs
        if d == 1 ? (not z.alcista and z.bot > close) : (z.alcista and z.top < close)
            dist = math.abs((d == 1 ? z.bot : z.top) - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                // Llegar al vacío = tocarlo. Pasarlo = cerrar más allá de todo el vacío.
                toque     := d == 1 ? z.bot : z.top
                umbral    := d == 1 ? z.top : z.bot
                txt       := (d == 1 ? "Vacío bajista " : "Vacío alcista ") + str.tostring(z.bot, format.mintick) + " – " + str.tostring(z.top, format.mintick)
    [toque, umbral, txt]

if nuevaSenal
    [tq, u, t] = buscarObjetivo(dir1h)
    objToque  := tq
    objUmbral := u
    objTxt    := t
    // Si en la misma vela de la señal el precio ya llegó al objetivo, la distribución ya terminó
    if not na(tq) and (dir1h == 1 ? high >= tq : low <= tq)
        origen    := "Objetivo alcanzado"
        dir1h     := 0
        objOrigen := na
        objToque  := na
        objUmbral := na
        objTxt    := ""

// =====================================================================
var table panel = table.new(position.top_right, 1, 5, border_width = 1)
if barstate.islast and esH1
    txt = dir1h == 1 ? "▲ COMPRA" : dir1h == -1 ? "▼ VENTA" : "Sin dirección"
    col = dir1h == 1 ? color.new(color.green, 10) : dir1h == -1 ? color.new(color.red, 10) : color.new(color.gray, 10)
    det = na(tSenal) ? "Esperando la primera señal" : origen + "\n" + str.format_time(tSenal, "dd/MM HH:mm", syminfo.timezone)
    table.cell(panel, 0, 0, "Dirección 1H", bgcolor = color.new(color.black, 20), text_color = color.white, text_size = size.small)
    table.cell(panel, 0, 1, txt, bgcolor = col, text_color = color.white, text_size = size.large)
    table.cell(panel, 0, 2, det, bgcolor = color.new(color.black, 20), text_color = color.white, text_size = size.small)
    table.cell(panel, 0, 3, dir1h != 0 ? "Objetivo: " + objTxt : "", bgcolor = color.new(color.black, 20), text_color = color.yellow, text_size = size.small)
    inval = dir1h != 0 and not na(objOrigen) ? "Se cancela si cierra " + (dir1h == 1 ? "por debajo de " : "por encima de ") + str.tostring(objOrigen, format.mintick) : ""
    table.cell(panel, 0, 4, inval, bgcolor = color.new(color.black, 20), text_color = color.orange, text_size = size.small)
