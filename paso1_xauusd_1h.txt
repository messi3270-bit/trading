//@version=6
indicator("Paso 1 · XAUUSD 1H · Manipulaciones, Vacíos y Order Blocks", shorttitle = "Paso 1 · 1H", overlay = true, max_boxes_count = 500, max_lines_count = 500, max_labels_count = 500)

// =====================================================================
// AJUSTES
// =====================================================================
grpN     = "Altos y bajos / Manipulaciones"
pivLen   = input.int(3, "Velas a cada lado para confirmar un alto/bajo", minval = 1, group = grpN)
maxVelas = input.int(4, "Velas máximas para volver dentro (manipulación)", minval = 1, group = grpN)
colNivel = input.color(color.gray, "Color altos/bajos sin tocar", group = grpN)
colManip = input.color(color.red, "Color manipulación", group = grpN)

grpF     = "Vacíos (FVG)"
showFvg  = input.bool(true, "Mostrar vacíos", group = grpF)
colFvgUp = input.color(color.new(#42A5F5, 80), "Vacío alcista", group = grpF)
colFvgDn = input.color(color.new(#1565C0, 80), "Vacío bajista", group = grpF)

grpO     = "Order Blocks"
showOb   = input.bool(true, "Mostrar order blocks", group = grpO)
colObUp  = input.color(color.new(#FF9800, 80), "OB alcista", group = grpO)
colObDn  = input.color(color.new(#E65100, 80), "OB bajista", group = grpO)

grpG     = "General"
maxZonas = input.int(15, "Máximo de vacíos / OB activos (de cada tipo)", minval = 1, group = grpG)
maxNivel = input.int(30, "Máximo de altos/bajos activos", minval = 2, group = grpG)

// =====================================================================
// TIPOS
// =====================================================================
// Un alto o un bajo de estructura que todavía no ha sido manipulado
type Nivel
    line  ln
    float precio
    bool  esAlto
    int   x1
    int   estado = 0      // 0 = esperando, 1 = una vela cerró fuera (contando velas)
    int   cuenta = 0      // velas pasadas desde que cerró fuera
    bool  vivo   = true

// Un vacío (FVG) o un order block
type Zona
    box   bx
    line  mid
    float top
    float bot
    bool  alcista
    bool  esFvg
    bool  tocado = false  // la vela anterior tocó la zona (50 % en FVG, cuerpo en OB)

var array<Nivel> niveles = array.new<Nivel>()
var array<Zona>  fvgs    = array.new<Zona>()
var array<Zona>  obs     = array.new<Zona>()

// =====================================================================
// FUNCIONES
// =====================================================================
nuevoNivel(float p, bool esAlto, int x) =>
    ln = line.new(x, p, bar_index, p, color = colNivel, style = line.style_dotted, extend = extend.right)
    Nivel.new(ln, p, esAlto, x)

nuevaZona(float top, float bot, bool alc, bool fvg, int x) =>
    c  = fvg ? (alc ? colFvgUp : colFvgDn) : (alc ? colObUp : colObDn)
    bx = box.new(x, top, bar_index, bot, border_color = color.new(c, 40), bgcolor = c, extend = extend.right, text = fvg ? "FVG" : "OB", text_size = size.tiny, text_color = color.new(c, 0), text_halign = text.align_left)
    line m = na
    if fvg
        // Línea del 50 % del vacío
        m := line.new(x, (top + bot) / 2, bar_index, (top + bot) / 2, color = color.new(c, 0), style = line.style_dashed, extend = extend.right)
    Zona.new(bx, m, top, bot, alc, fvg)

borrarZona(Zona z) =>
    z.bx.delete()
    if not na(z.mid)
        z.mid.delete()

// Revisa cada zona activa: invalidación y reacción
procesarZonas(array<Zona> arr) =>
    if arr.size() > 0
        for i = arr.size() - 1 to 0
            z   = arr.get(i)
            mid = (z.top + z.bot) / 2
            // FVG: el precio tiene que llegar al 50 % o más. OB: basta con tocar el cuerpo.
            toca = z.esFvg ? (z.alcista ? low <= mid : high >= mid) : (z.alcista ? low <= z.top : high >= z.bot)
            // Invalidación: una vela cierra al otro lado de la zona
            invalida = z.alcista ? close < z.bot : close > z.top
            // Reacción: la vela anterior tocó y esta vela va a favor
            reacciona = z.tocado and (z.alcista ? close > open : close < open)
            borrar = false
            if invalida
                borrar := true
            else if reacciona
                // Solo vale una reacción: se marca con ✓ y se borra la zona
                c = z.esFvg ? (z.alcista ? colFvgUp : colFvgDn) : (z.alcista ? colObUp : colObDn)
                label.new(bar_index, z.alcista ? low : high, "✓ " + (z.esFvg ? "FVG" : "OB"), style = z.alcista ? label.style_label_up : label.style_label_down, color = color.new(c, 0), textcolor = color.white, size = size.tiny)
                borrar := true
            else
                z.tocado := toca
            if borrar
                borrarZona(z)
                arr.remove(i)

limitar(array<Zona> arr) =>
    while arr.size() > maxZonas
        borrarZona(arr.shift())

// =====================================================================
// 1) MANIPULACIONES SOBRE LOS ALTOS Y BAJOS ACTIVOS
// =====================================================================
if niveles.size() > 0
    for i = niveles.size() - 1 to 0
        n = niveles.get(i)
        if not n.vivo
            niveles.remove(i)
            continue
        manip      = false
        roto       = false
        cerroFuera = false
        // ¿La vela cierra dentro del nivel (por debajo del alto / por encima del bajo)?
        dentro = n.esAlto ? close <= n.precio : close >= n.precio
        if n.estado == 0
            // Pasa el nivel: si cierra dentro es manipulación con mecha; si cierra fuera, empieza la cuenta
            pasa = n.esAlto ? high > n.precio : low < n.precio
            manip      := pasa and dentro
            cerroFuera := pasa and not dentro
        else
            // Cerró fuera: tiene como máximo "maxVelas" velas para volver a cerrar dentro
            n.cuenta := n.cuenta + 1
            manip    := dentro
            roto     := not dentro and n.cuenta >= maxVelas
        if cerroFuera
            n.estado := 1
        if manip
            n.ln.set_extend(extend.none)
            n.ln.set_x2(bar_index)
            n.ln.set_color(colManip)
            n.ln.set_style(line.style_solid)
            n.ln.set_width(2)
            label.new(bar_index, n.precio, "M", style = n.esAlto ? label.style_label_down : label.style_label_up, color = colManip, textcolor = color.white, size = size.small)
            n.vivo := false
            niveles.remove(i)
        else if roto
            // Rompimiento real: no se marca, solo se quita el nivel
            n.ln.delete()
            n.vivo := false
            niveles.remove(i)

// =====================================================================
// 2) NUEVOS ALTOS Y BAJOS (siempre alternados: alto → bajo → alto → bajo)
// =====================================================================
ph = ta.pivothigh(high, pivLen, pivLen)
pl = ta.pivotlow(low, pivLen, pivLen)

var int   ultimoTipo = 0     // 1 = el último fue un alto, -1 = el último fue un bajo
var Nivel ultAlto    = na
var Nivel ultBajo    = na

if not na(ph)
    // Dos altos seguidos sin bajo en medio: se queda solo el más alto
    repetido = ultimoTipo == 1
    agregar  = not repetido or ph > ultAlto.precio
    if agregar
        if repetido and ultAlto.vivo and ultAlto.estado == 0
            ultAlto.ln.delete()
            ultAlto.vivo := false
        ultAlto := nuevoNivel(ph, true, bar_index - pivLen)
        niveles.push(ultAlto)
        ultimoTipo := 1

if not na(pl)
    // Dos bajos seguidos sin alto en medio: se queda solo el más bajo
    repetido = ultimoTipo == -1
    agregar  = not repetido or pl < ultBajo.precio
    if agregar
        if repetido and ultBajo.vivo and ultBajo.estado == 0
            ultBajo.ln.delete()
            ultBajo.vivo := false
        ultBajo := nuevoNivel(pl, false, bar_index - pivLen)
        niveles.push(ultBajo)
        ultimoTipo := -1

// Limita la cantidad de altos/bajos en pantalla
while niveles.size() > maxNivel
    viejo = niveles.shift()
    if viejo.vivo
        viejo.ln.delete()
        viejo.vivo := false

// =====================================================================
// 3) VACÍOS Y ORDER BLOCKS: primero se revisan los que ya existen...
// =====================================================================
procesarZonas(fvgs)
procesarZonas(obs)

// ...y luego se buscan los nuevos (así empiezan a contar desde la vela siguiente)

// Vacíos de 3 velas
if showFvg
    if low > high[2]
        fvgs.push(nuevaZona(low, high[2], true, true, bar_index - 2))
    if high < low[2]
        fvgs.push(nuevaZona(low[2], high, false, true, bar_index - 2))
    limitar(fvgs)

// Order blocks: última vela contraria antes del movimiento
var int   rojaIdx  = na
var float rojaO    = na
var float rojaC    = na
var float rojaH    = na
var int   verdeIdx = na
var float verdeO   = na
var float verdeC   = na
var float verdeL   = na

// OB alcista: una vela cierra por encima del máximo de la última vela roja
if not na(rojaIdx) and close > rojaH
    if showOb
        obs.push(nuevaZona(rojaO, rojaC, true, false, rojaIdx))
    rojaIdx := na

// OB bajista: una vela cierra por debajo del mínimo de la última vela verde
if not na(verdeIdx) and close < verdeL
    if showOb
        obs.push(nuevaZona(verdeC, verdeO, false, false, verdeIdx))
    verdeIdx := na

limitar(obs)

// Guarda la última vela roja y la última vela verde
if close < open
    rojaIdx := bar_index
    rojaO   := open
    rojaC   := close
    rojaH   := high
if close > open
    verdeIdx := bar_index
    verdeO   := open
    verdeC   := close
    verdeL   := low

// =====================================================================
// AVISO SI NO ESTÁS EN 1H
// =====================================================================
var table aviso = table.new(position.top_right, 1, 1)
if barstate.islast and timeframe.period != "60"
    table.cell(aviso, 0, 0, "⚠ Este indicador está pensado para 1H", bgcolor = color.new(color.orange, 20), text_color = color.white)
