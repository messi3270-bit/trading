// Un alto o un bajo de estructura que todavía no ha sido manipulado
type Nivel
    line  ln
    float precio
    bool  esAlto
    int   x1
    int   estado = 0      // 0 = esperando, 1 = una vela cerró fuera (contando velas)
    int   cuenta = 0      // velas pasadas desde que cerró fuera
    float extremo = na    // punta máxima (alto) o mínima (bajo) a la que llegó la manipulación
    bool  vivo   = true

// Un vacío (FVG) o un order block
type Zona
    box   bx
    line  mid
    float top
    float bot
    bool  alcista
    bool  esFvg
    bool  tocado = false  // la vela anterior tocó la zona (50 % en FVG, cuerpo en OB)
    int   dentro = 0      // velas que han tocado la zona desde que se creó, aunque no sean seguidas (solo OB)
    color col    = na     // color de la zona (para poder ocultarla y volver a mostrarla)

// =====================================================================
// MOTOR DE LA DIRECCIÓN DE 1H (sin dibujos)
// Es una copia exacta de las reglas del paso 1, pero sin dibujar nada,
// para poder calcular la dirección de 1H desde 5 minutos (o cualquier temporalidad menor).
// =====================================================================
type NivelP
    float precio
    bool  esAlto
    int   estado  = 0
    int   cuenta  = 0
    float extremo = na
    bool  vivo    = true

type ZonaP
    float top
    float bot
    bool  alcista
    bool  esFvg
    bool  tocado = false
    int   dentro = 0

procZonasP(array<ZonaP> arr) =>
    int   dir     = 0
    float borde   = na
    int   ruptura = 0
    if arr.size() > 0
        for i = arr.size() - 1 to 0
            z    = arr.get(i)
            mid  = (z.top + z.bot) / 2
            toca = z.esFvg ? (z.alcista ? low <= mid : high >= mid) : (z.alcista ? low <= z.top : high >= z.bot)
            invalida = z.alcista ? close < z.bot : close > z.top
            enZona   = low <= z.top and high >= z.bot
            z.dentro := enZona ? z.dentro + 1 : z.dentro
            atascado = not z.esFvg and z.dentro > maxDentro
            aFavor    = z.alcista ? close > open : close < open
            reacciona = aFavor and (toca or z.tocado)
            borrar = false
            if invalida or atascado
                if invalida
                    ruptura := z.alcista ? -1 : 1
                borrar := true
            else if reacciona
                dir    := z.alcista ? 1 : -1
                borde  := z.alcista ? z.bot : z.top
                borrar := true
            else
                z.tocado := toca
            if borrar
                arr.remove(i)
    [dir, borde, ruptura]

objetivoP(int d, array<NivelP> nv, array<ZonaP> ob, array<ZonaP> fv) =>
    float mejorDist = na
    float toque     = na
    float umbral    = na
    for n in nv
        if n.vivo and (d == 1 ? (n.esAlto and n.precio > close) : (not n.esAlto and n.precio < close))
            dist = math.abs(n.precio - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                toque     := n.precio
                umbral    := n.precio
    for z in ob
        if d == 1 ? (not z.alcista and z.bot > close) : (z.alcista and z.top < close)
            dist = math.abs((d == 1 ? z.bot : z.top) - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                toque     := d == 1 ? z.bot : z.top
                umbral    := d == 1 ? z.top : z.bot
    for z in fv
        if d == 1 ? (not z.alcista and z.bot > close) : (z.alcista and z.top < close)
            dist = math.abs((d == 1 ? z.bot : z.top) - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                toque     := d == 1 ? z.bot : z.top
                umbral    := d == 1 ? z.top : z.bot
    [toque, umbral]

// Devuelve la foto de la VELA ANTERIOR (cerrada) de esa temporalidad:
// [dirección, techos, suelos, ¿alcista?, ¿es vacío?, manipulaciones confirmadas (nivel), ¿era un alto?, dirección de la vela ACTUAL]
// dirección: 1 = compra, -1 = venta, 0 = sin dirección
motor() =>
    var array<NivelP> nv = array.new<NivelP>()
    var array<ZonaP>  fv = array.new<ZonaP>()
    var array<ZonaP>  ob = array.new<ZonaP>()
    var int    dir     = 0
    var float  oOrigen = na
    var float  oToque  = na
    var float  oUmbral = na
    var int    ultTipo = 0
    var NivelP uA      = na
    var NivelP uB      = na
    var int    rIdx    = na
    var float  rO      = na
    var float  rC      = na
    var int    vIdx    = na
    var float  vO      = na
    var float  vC      = na
    nueva = false
    // Manipulaciones confirmadas en esta vela (nivel y si era un alto)
    cNiv  = array.new<float>()
    cAlto = array.new<bool>()
    // Ruptura del origen: cambia de dirección. Ruptura del objetivo a favor: continuación.
    if dir != 0 and not na(oOrigen) and (dir == 1 ? close < oOrigen : close > oOrigen)
        dir     := -dir
        oOrigen := dir == 1 ? low : high
        nueva   := true
    else if dir != 0 and not na(oUmbral) and (dir == 1 ? close > oUmbral : close < oUmbral)
        oOrigen := dir == 1 ? low : high
        nueva   := true
    libre = dir == 0 or na(oUmbral)
    // Manipulaciones
    if nv.size() > 0
        for i = nv.size() - 1 to 0
            n = nv.get(i)
            if not n.vivo
                nv.remove(i)
                continue
            manip = false
            roto  = false
            cf    = false
            dentro = n.esAlto ? close <= n.precio : close >= n.precio
            if n.estado == 0
                pasa = n.esAlto ? high > n.precio : low < n.precio
                if pasa
                    n.extremo := n.esAlto ? high : low
                manip := pasa and dentro
                cf    := pasa and not dentro
            else
                n.cuenta  := n.cuenta + 1
                n.extremo := n.esAlto ? math.max(n.extremo, high) : math.min(n.extremo, low)
                manip := dentro
                roto  := not dentro and n.cuenta >= maxVelas
            // Si el precio se alejó demasiado del nivel, es un rompimiento y no una manipulación
            if manip and not na(n.extremo) and math.abs(n.extremo - n.precio) > maxDist
                manip := false
                roto  := true
            if cf
                n.estado := 1
                // Ruptura del nivel: dirección hacia donde rompe
                dR = n.esAlto ? 1 : -1
                if libre or dR == -dir
                    dir     := dR
                    oOrigen := dR == 1 ? low : high
                    nueva   := true
                    libre   := false
                else if dR == dir
                    oOrigen := dR == 1 ? low : high
                    nueva   := true
            if manip
                cNiv.push(n.precio)
                cAlto.push(n.esAlto)
            if manip or roto
                dM = n.esAlto ? -1 : 1
                if manip and (libre or dM == -dir)
                    dir     := dM
                    oOrigen := n.extremo
                    nueva   := true
                    libre   := false
                n.vivo := false
                nv.remove(i)
    // Altos y bajos alternados
    ph = ta.pivothigh(high, pivLen, pivLen)
    pl = ta.pivotlow(low, pivLen, pivLen)
    if not na(ph)
        rep = ultTipo == 1
        if not rep or ph > uA.precio
            if rep and uA.vivo and uA.estado == 0
                uA.vivo := false
            uA := NivelP.new(ph, true)
            nv.push(uA)
            ultTipo := 1
    if not na(pl)
        rep = ultTipo == -1
        if not rep or pl < uB.precio
            if rep and uB.vivo and uB.estado == 0
                uB.vivo := false
            uB := NivelP.new(pl, false)
            nv.push(uB)
            ultTipo := -1
    while nv.size() > maxNivel
        v = nv.shift()
        v.vivo := false
    // Reacciones en vacíos y OB
    [dF, bF, rupF] = procZonasP(fv)
    if dF != 0 and (libre or dF == -dir)
        dir     := dF
        oOrigen := bF
        nueva   := true
        libre   := false
    if rupF != 0
        if libre or rupF == -dir
            dir     := rupF
            oOrigen := rupF == 1 ? low : high
            nueva   := true
            libre   := false
        else if rupF == dir
            oOrigen := rupF == 1 ? low : high
            nueva   := true
    [dO, bO, rupO] = procZonasP(ob)
    if dO != 0 and (libre or dO == -dir)
        dir     := dO
        oOrigen := bO
        nueva   := true
        libre   := false
    if rupO != 0
        if libre or rupO == -dir
            dir     := rupO
            oOrigen := rupO == 1 ? low : high
            nueva   := true
            libre   := false
        else if rupO == dir
            oOrigen := rupO == 1 ? low : high
            nueva   := true
    // Vacíos nuevos
    if showFvg
        if low > high[2]
            fv.push(ZonaP.new(low, high[2], true, true))
        if high < low[2]
            fv.push(ZonaP.new(low[2], high, false, true))
    while fv.size() > maxZonas
        fv.shift()
    // OB nuevos
    if not na(rIdx) and close > rO
        if showOb
            ob.push(ZonaP.new(rO, rC, true, false))
        rIdx := na
    if not na(vIdx) and close < vO
        if showOb
            ob.push(ZonaP.new(vC, vO, false, false))
        vIdx := na
    while ob.size() > maxZonas
        ob.shift()
    if close < open
        rIdx := bar_index
        rO   := open
        rC   := close
    if close > open
        vIdx := bar_index
        vO   := open
        vC   := close
    // Objetivo de la dirección nueva
    if nueva
        [tq, u] = objetivoP(dir, nv, ob, fv)
        oToque  := tq
        oUmbral := u
    // Foto de la vela anterior: así, con lookahead, no repinta
    var int          pDir  = 0
    var array<float> pTop  = array.new<float>()
    var array<float> pBot  = array.new<float>()
    var array<bool>  pAlc  = array.new<bool>()
    var array<bool>  pFvg  = array.new<bool>()
    var array<float> pNiv  = array.new<float>()
    var array<bool>  pAlto = array.new<bool>()
    oDir  = pDir
    oTop  = pTop
    oBot  = pBot
    oAlc  = pAlc
    oFvg  = pFvg
    oNiv  = pNiv
    oAlto = pAlto
    cTop  = array.new<float>()
    cBot  = array.new<float>()
    cAlc  = array.new<bool>()
    cFvg  = array.new<bool>()
    for z in fv
        cTop.push(z.top)
        cBot.push(z.bot)
        cAlc.push(z.alcista)
        cFvg.push(true)
    for z in ob
        cTop.push(z.top)
        cBot.push(z.bot)
        cAlc.push(z.alcista)
        cFvg.push(false)
    pDir  := dir
    pTop  := cTop
    pBot  := cBot
    pAlc  := cAlc
    pFvg  := cFvg
    pNiv  := cNiv
    pAlto := cAlto
    [oDir, oTop, oBot, oAlc, oFvg, oNiv, oAlto, dir]

// Muestra u oculta una zona (oculta = colores transparentes)
mostrarZona(Zona z, bool vis) =>
    z.bx.set_bgcolor(vis ? z.col : color.new(z.col, 100))

    z.bx.set_text_color(color.new(z.col, vis ? 0 : 100))
    if not na(z.mid)
        z.mid.set_color(color.new(z.col, vis ? 0 : 100))

nuevaZona(float top, float bot, bool alc, bool fvg, int x) =>
    c  = fvg ? (alc ? colFvgUp : colFvgDn) : (alc ? colObUp : colObDn)
    bx = box.new(x, top, bar_index, bot, border_color = color.new(c, 40), bgcolor = c, extend = extend.right, text = fvg ? "FVG" : "OB", text_size = size.tiny, text_color = color.new(c, 0), text_halign = text.align_left)
    line m = na
    if fvg
        // Línea del 50 % del vacío
        m := line.new(x, (top + bot) / 2, bar_index, (top + bot) / 2, color = color.new(c, 0), style = line.style_dashed, extend = extend.right)
    z = Zona.new(bx, m, top, bot, alc, fvg, col = c)
    if not zonaVisible(alc)
        mostrarZona(z, false)
    z

borrarZona(Zona z) =>
    z.bx.delete()
    if not na(z.mid)
        z.mid.delete()

// Revisa cada zona activa: invalidación y reacción.
// Devuelve [dirección, borde de origen, ruptura]: 1 = compra, -1 = venta, 0 = ninguna.
// El borde de origen es el lado de la zona que, si se cierra más allá, cambia la dirección.
// Ruptura: una vela cierra al otro lado de la zona (por debajo de una alcista = venta; por encima de una bajista = compra).
procesarZonas(array<Zona> arr) =>
    int   dir     = 0
    float borde   = na
    int   ruptura = 0
    if arr.size() > 0
        for i = arr.size() - 1 to 0
            z   = arr.get(i)
            mid = (z.top + z.bot) / 2
            // FVG: el precio tiene que llegar al 50 % o más. OB: basta con tocar el cuerpo.
            toca = z.esFvg ? (z.alcista ? low <= mid : high >= mid) : (z.alcista ? low <= z.top : high >= z.bot)
            // Invalidación: una vela cierra al otro lado de la zona
            invalida = z.alcista ? close < z.bot : close > z.top
            // OB: si más de "maxDentro" velas han tocado el cuerpo (aunque no sean seguidas) = consolidación: lo va a romper, se invalida
            enZona   = low <= z.top and high >= z.bot
            z.dentro := enZona ? z.dentro + 1 : z.dentro
            atascado = not z.esFvg and z.dentro > maxDentro
            // Reacción: la vela que toca cierra a favor, o la vela anterior tocó y esta cierra a favor
            aFavor    = z.alcista ? close > open : close < open
            reacciona = aFavor and (toca or z.tocado)
            borrar = false
            if invalida or atascado
                if invalida
                    ruptura := z.alcista ? -1 : 1
                borrar := true
            else if reacciona
                // Solo vale una reacción: se marca con ✓ y se borra la zona
                c = z.esFvg ? (z.alcista ? colFvgUp : colFvgDn) : (z.alcista ? colObUp : colObDn)
                if zonaVisible(z.alcista)
                    label.new(bar_index, z.alcista ? low : high, "✓ " + (z.esFvg ? "FVG" : "OB"), style = z.alcista ? label.style_label_up : label.style_label_down, color = color.new(c, 0), textcolor = color.white, size = size.tiny)
                dir    := z.alcista ? 1 : -1
                borde  := z.alcista ? z.bot : z.top
                borrar := true
            else
                z.tocado := toca
            if borrar
                borrarZona(z)
                arr.remove(i)
    [dir, borde, ruptura]

limitar(array<Zona> arr) =>
    while arr.size() > maxZonas
        borrarZona(arr.shift())

// =====================================================================
// DIRECCIÓN DE 1H
// Señales: reacciones (✓), manipulaciones (M) y RUPTURAS (una vela cierra al otro lado de un alto/bajo, OB o vacío).
// Una señal CONTRARIA cambia la dirección. Una RUPTURA a favor es continuación: se busca el siguiente objetivo.
// Si una vela cierra más allá del OBJETIVO a favor: continuación hacia el siguiente objetivo.
// Si una vela cierra más allá del ORIGEN de la señal: cambia de dirección.
// Si el precio solo TOCA el objetivo, la dirección sigue: si reacciona ahí, la reacción la cambia.
// =====================================================================
var int    dir1h    = 0     // 1 = compra, -1 = venta, 0 = todavía sin señal
var string origen   = ""
var int    tSenal   = na
var float  objOrigen = na   // si una vela de 1H cierra más allá del origen de la señal (en contra), se cancela
var float  objToque  = na   // precio donde el precio LLEGA al objetivo
var float  objUmbral = na   // si una vela de 1H CIERRA más allá de este precio, la dirección se cancela
var string objTxt    = ""
nuevaSenal = false

// Ruptura del ORIGEN: el precio cierra más allá del punto donde nació la señal → cambia de dirección
if dir1h != 0 and not na(objOrigen) and (dir1h == 1 ? close < objOrigen : close > objOrigen)
    dir1h      := -dir1h
    origen     := "Rompió el origen: cambia a " + (dir1h == 1 ? "compra" : "venta")
    tSenal     := time
    objOrigen  := dir1h == 1 ? low : high
    nuevaSenal := true
// Ruptura del OBJETIVO a favor → continuación: se busca el siguiente objetivo
else if dir1h != 0 and not na(objUmbral) and (dir1h == 1 ? close > objUmbral : close < objUmbral)
    origen     := "Rompió el objetivo: continuación"
    tSenal     := time
    objOrigen  := dir1h == 1 ? low : high
    nuevaSenal := true

// Solo se acepta una señal nueva si no hay dirección activa (o si la activa no tiene objetivo)
libre = dir1h == 0 or na(objUmbral)

// =====================================================================
// 1) MANIPULACIONES SOBRE LOS ALTOS Y BAJOS ACTIVOS
// =====================================================================
if niveles.size() > 0
    for i = niveles.size() - 1 to 0
        n = niveles.get(i)
        if not n.vivo
            niveles.remove(i)
            continue
        manip      = false
        roto       = false
        cerroFuera = false
        // ¿La vela cierra dentro del nivel (por debajo del alto / por encima del bajo)?
        dentro = n.esAlto ? close <= n.precio : close >= n.precio
        if n.estado == 0
            // Pasa el nivel: si cierra dentro es manipulación con mecha; si cierra fuera, empieza la cuenta
            pasa = n.esAlto ? high > n.precio : low < n.precio
            if pasa
                n.extremo := n.esAlto ? high : low
            manip      := pasa and dentro
            cerroFuera := pasa and not dentro
        else
            // Cerró fuera: tiene como máximo "maxVelas" velas para volver a cerrar dentro
            n.cuenta  := n.cuenta + 1
            n.extremo := n.esAlto ? math.max(n.extremo, high) : math.min(n.extremo, low)
            manip    := dentro
            roto     := not dentro and n.cuenta >= maxVelas
        // Si el precio se alejó demasiado del nivel, es un rompimiento y no una manipulación
        if manip and not na(n.extremo) and math.abs(n.extremo - n.precio) > maxDist
            manip := false
            roto  := true
        if cerroFuera
            n.estado := 1
            // Ruptura del nivel: la vela cierra fuera → dirección hacia donde rompe
            // (si vuelve a cerrar dentro en 4 velas, será manipulación y cambiará al revés)
            dR = n.esAlto ? 1 : -1
            if libre or dR == -dir1h
                dir1h      := dR
                origen     := n.esAlto ? "Ruptura de alto" : "Ruptura de bajo"
                tSenal     := time
                objOrigen  := dR == 1 ? low : high
                nuevaSenal := true
                libre      := false
            else if dR == dir1h
                origen     := (n.esAlto ? "Ruptura de alto" : "Ruptura de bajo") + " (continuación)"
                tSenal     := time
                objOrigen  := dR == 1 ? low : high
                nuevaSenal := true
        if manip
            if nivelVisible(n.esAlto)
                label.new(bar_index, n.precio, "M", style = n.esAlto ? label.style_label_down : label.style_label_up, color = colManip, textcolor = color.white, size = size.small)
                n.ln.set_extend(extend.none)
                n.ln.set_x2(bar_index)
                n.ln.set_color(colManip)
                n.ln.set_style(line.style_solid)
                n.ln.set_width(2)
            else
                n.ln.delete()
            // Manipulación de un alto = venta. Manipulación de un bajo = compra.
            // Una manipulación contraria anula la dirección actual y pone la nueva.
            dirM = n.esAlto ? -1 : 1
            if libre or dirM == -dir1h
                dir1h      := n.esAlto ? -1 : 1
                origen     := n.esAlto ? "Manipulación de alto" : "Manipulación de bajo"
                tSenal     := time
                objOrigen  := n.extremo
                nuevaSenal := true
                libre      := false
            n.vivo := false
            niveles.remove(i)
        else if roto
            // Rompimiento real: no se marca, solo se quita el nivel
            n.ln.delete()
            n.vivo := false
            niveles.remove(i)

// =====================================================================
// 2) NUEVOS ALTOS Y BAJOS (siempre alternados: alto → bajo → alto → bajo)
// =====================================================================
ph = ta.pivothigh(high, pivLen, pivLen)
pl = ta.pivotlow(low, pivLen, pivLen)

var int   ultimoTipo = 0     // 1 = el último fue un alto, -1 = el último fue un bajo
var Nivel ultAlto    = na
var Nivel ultBajo    = na

if activo and not na(ph)
    // Dos altos seguidos sin bajo en medio: se queda solo el más alto
    repetido = ultimoTipo == 1
    agregar  = not repetido or ph > ultAlto.precio
    if agregar
        if repetido and ultAlto.vivo and ultAlto.estado == 0
            ultAlto.ln.delete()
            ultAlto.vivo := false
        ultAlto := nuevoNivel(ph, true, bar_index - pivLen)
        niveles.push(ultAlto)
        ultimoTipo := 1

if activo and not na(pl)
    // Dos bajos seguidos sin alto en medio: se queda solo el más bajo
    repetido = ultimoTipo == -1
    agregar  = not repetido or pl < ultBajo.precio
    if agregar
        if repetido and ultBajo.vivo and ultBajo.estado == 0
            ultBajo.ln.delete()
            ultBajo.vivo := false
        ultBajo := nuevoNivel(pl, false, bar_index - pivLen)
        niveles.push(ultBajo)
        ultimoTipo := -1

// Limita la cantidad de altos/bajos en pantalla
while niveles.size() > maxNivel
    viejo = niveles.shift()
    if viejo.vivo
        viejo.ln.delete()
        viejo.vivo := false

// =====================================================================
// PASO 2: cuando cambia la dirección de 1H, se muestran solo las zonas y niveles a favor
// =====================================================================
if esMenor and dirHTF != dirHTF[1]
    for z in fvgs
        mostrarZona(z, zonaVisible(z.alcista))
    for z in obs
        mostrarZona(z, zonaVisible(z.alcista))
    for n in niveles
        n.ln.set_color(nivelVisible(n.esAlto) ? colNivel : transp)

// =====================================================================
// 3) VACÍOS Y ORDER BLOCKS: primero se revisan los que ya existen...
// =====================================================================
[dirFvg, bordeFvg, rupFvg] = procesarZonas(fvgs)
// Una señal CONTRARIA (vacío u OB) anula la dirección actual: empieza la distribución hacia el otro lado
if dirFvg != 0 and (libre or dirFvg == -dir1h)
    dir1h      := dirFvg
    origen     := dirFvg == 1 ? "Reacción en vacío alcista" : "Reacción en vacío bajista"
    tSenal     := time
    objOrigen  := bordeFvg
    nuevaSenal := true
    libre      := false
// Ruptura de un vacío: cierre al otro lado → dirección hacia donde rompe
if rupFvg != 0
    if libre or rupFvg == -dir1h
        dir1h      := rupFvg
        origen     := rupFvg == 1 ? "Ruptura de vacío bajista" : "Ruptura de vacío alcista"
        tSenal     := time
        objOrigen  := rupFvg == 1 ? low : high
        nuevaSenal := true
        libre      := false
    else if rupFvg == dir1h
        origen     := (rupFvg == 1 ? "Ruptura de vacío bajista" : "Ruptura de vacío alcista") + " (continuación)"
        tSenal     := time
        objOrigen  := rupFvg == 1 ? low : high
        nuevaSenal := true
[dirOb, bordeOb, rupOb] = procesarZonas(obs)
if dirOb != 0 and (libre or dirOb == -dir1h)
    dir1h      := dirOb
    origen     := dirOb == 1 ? "Reacción en OB alcista" : "Reacción en OB bajista"
    tSenal     := time
    objOrigen  := bordeOb
    nuevaSenal := true
    libre      := false
// Ruptura de un OB: cierre al otro lado del cuerpo → dirección hacia donde rompe
if rupOb != 0
    if libre or rupOb == -dir1h
        dir1h      := rupOb
        origen     := rupOb == 1 ? "Ruptura de OB bajista" : "Ruptura de OB alcista"
        tSenal     := time
        objOrigen  := rupOb == 1 ? low : high
        nuevaSenal := true
        libre      := false
    else if rupOb == dir1h
        origen     := (rupOb == 1 ? "Ruptura de OB bajista" : "Ruptura de OB alcista") + " (continuación)"
        tSenal     := time
        objOrigen  := rupOb == 1 ? low : high
        nuevaSenal := true

// ...y luego se buscan los nuevos (así empiezan a contar desde la vela siguiente)

// =====================================================================
// PASO 3 (1 MINUTO): BÚSQUEDA Y ENTRADAS
// =====================================================================
// Vela de 5 minutos construida con las velas de 1 minuto (para saber cuándo se rompe la zona de 5m)
var float v5h = na
var float v5l = na
var float v5c = na
nueva5 = timeframe.change("5")
entradaCompra = false
entradaVenta  = false
// Bloqueo: el precio tocó un OB de 1H CONTRARIO a la dirección (OB bajista en compra, OB alcista en venta)
// y ese OB todavía no se ha decidido (ni ✓ ni ruptura en 1H). Mientras tanto no se busca ninguna entrada.
var array<string> tocadas1h = array.new<string>()
bloqueo1h = false
string bloqueoTxt = ""
int   nOb1hContra = 0
float distOb1h    = na
string obCercaTxt = "ninguno"

// Toques de zonas de 5m de la dirección CONTRARIA a 1H. Si 1H cambia a esa dirección
// dentro de la misma vela de 1H, el toque cuenta como requisito de 5m.
var array<Busqueda> pendientes = array.new<Busqueda>()

claveB(Busqueda b) =>
    b.tipo == 2 ? claveZona(2, b.nivel, b.nivel) : claveZona(b.tipo, b.top, b.bot)

estaPendiente(string k) =>
    r = false
    for b in pendientes
        if claveB(b) == k
            r := true
    r

// ¿Se rompió la zona de 5m al cerrar la vela de 5m? (marca b.viva = false)
evalRotura(Busqueda b) =>
    if b.tipo == 0
        // OB: una vela de 5m cierra al otro lado, o más de 4 velas de 5m lo han tocado (aunque no sean seguidas)
        enZona5   = v5l <= b.top and v5h >= b.bot
        b.dentro5 := enZona5 ? b.dentro5 + 1 : b.dentro5
        if (b.dir == 1 ? v5c < b.bot : v5c > b.top) or b.dentro5 > maxDentro
            b.viva := false
    else if b.tipo == 1
        // Vacío: una vela de 5m cierra al otro lado
        if b.dir == 1 ? v5c < b.bot : v5c > b.top
            b.viva := false
    else
        // Manipulación: cierra fuera del nivel y no vuelve dentro en 4 velas de 5m = rompimiento real
        b.fuera5 := (b.dir == 1 ? v5c < b.nivel : v5c > b.nivel) ? b.fuera5 + 1 : 0
        if b.fuera5 > maxVelas
            b.viva := false

if esM1
    // 0) Si cambia la dirección de 1H: las búsquedas de entrada de la dirección anterior se reinician.
    //    Las operaciones ya abiertas NO se tocan: siguen hasta su TP o su SL.
    if dirHTF != dirHTF[1]
        busquedas.clear()
        limpiarObs1()
        logE("1H cambia a " + (dirHTF == 1 ? "COMPRA" : dirHTF == -1 ? "VENTA" : "sin dirección"))
        // Requisito de 5m que ya ocurrió dentro de la vela de 1H que dio la señal (o la que está en curso)
        if dirHTF != 0 and abiertas.size() == 0
            tMin = time("60") - 3600000
            int idx = -1
            if pendientes.size() > 0
                for i = 0 to pendientes.size() - 1
                    pb = pendientes.get(i)
                    if pb.dir == dirHTF and pb.viva and pb.tInicio >= tMin
                        idx := i
            if idx >= 0
                pb = pendientes.get(idx)
                usadas.push(claveB(pb))
                busquedas.push(pb)
                logE("Requisito 5m previo: " + descB(pb))
        pendientes.clear()

    // 1) Al cerrar cada vela de 5m: ¿se rompió la zona de 5m de alguna búsqueda?
    if nueva5 and not na(v5c)
        for b in busquedas
            evalRotura(b)
        for b in pendientes
            evalRotura(b)
    // Los toques pendientes solo valen si siguen vivos y son de las últimas 2 horas
    if pendientes.size() > 0
        for i = pendientes.size() - 1 to 0
            pb = pendientes.get(i)
            if not pb.viva or pb.tInicio < time - 7200000
                pendientes.remove(i)
    if busquedas.size() > 0
        for i = busquedas.size() - 1 to 0
            if not busquedas.get(i).viva
                logE("Zona rota, fin búsqueda: " + descB(busquedas.get(i)))
                busquedas.remove(i)

    // 2) El stop loss sigue el mínimo (compra) o máximo (venta) desde el toque
    for b in busquedas
        b.extremo := b.dir == 1 ? math.min(b.extremo, low) : math.max(b.extremo, high)

    // 2b) Bloqueo por OB de 1H contrario: hay que esperar a ver si lo respeta o lo rompe
    if dirHTF != 0 and not na(z60Top)
        if z60Top.size() > 0
            for i = 0 to z60Top.size() - 1
                // Solo OB (no vacíos) y solo los contrarios a la dirección de 1H
                if not z60Fvg.get(i) and z60Alc.get(i) == (dirHTF == -1)
                    t1 = z60Top.get(i)
                    b1 = z60Bot.get(i)
                    k1 = claveZona(9, t1, b1)
                    nOb1hContra := nOb1hContra + 1
                    dist1 = close > t1 ? close - t1 : close < b1 ? b1 - close : 0.0
                    if na(distOb1h) or dist1 < distOb1h
                        distOb1h   := dist1
                        obCercaTxt := str.tostring(b1, format.mintick) + "–" + str.tostring(t1, format.mintick) + (tocadas1h.includes(k1) ? " (tocado)" : "")
                    if low <= t1 and high >= b1 and not tocadas1h.includes(k1)
                        tocadas1h.push(k1)
                    if tocadas1h.includes(k1)
                        bloqueo1h  := true
                        bloqueoTxt := str.tostring(b1, format.mintick) + " – " + str.tostring(t1, format.mintick)
    while tocadas1h.size() > 100
        tocadas1h.shift()
    if bloqueo1h and not bloqueo1h[1]
        logE("Bloqueo: precio en OB de 1H " + bloqueoTxt)
    if not bloqueo1h and bloqueo1h[1]
        logE("Fin del bloqueo por OB de 1H")
    if bloqueo1h
        busquedas.clear()
        limpiarObs1()

    // 3) Si no hay ninguna búsqueda activa: ¿el precio toca una zona de 5m a favor de 1H que no se haya usado?
    //    Solo hay UNA búsqueda a la vez. Cada zona de 5m se usa una sola vez.
    //    Mientras haya una operación abierta (sin tocar TP ni SL) no se busca nada.
    if not bloqueo1h and abiertas.size() == 0 and busquedas.size() == 0 and dirHTF != 0 and not na(z5Top)
        if z5Top.size() > 0
            for i = 0 to z5Top.size() - 1
                alc = z5Alc.get(i)
                if alc == (dirHTF == 1) and busquedas.size() == 0
                    t   = z5Top.get(i)
                    bt  = z5Bot.get(i)
                    fvg = z5Fvg.get(i)
                    mid = (t + bt) / 2
                    // OB: toca el cuerpo. Vacío: llega al 50 %.
                    toca  = fvg ? (alc ? low <= mid : high >= mid) : (alc ? low <= t : high >= bt)
                    clave = claveZona(fvg ? 1 : 0, t, bt)
                    if toca and not usadas.includes(clave)
                        usadas.push(clave)
                        busquedas.push(Busqueda.new(dirHTF, fvg ? 1 : 0, t, bt, na, alc ? low : high, time))
                        logE("Toca zona: " + descB(busquedas.get(0)))
    if not bloqueo1h and abiertas.size() == 0 and busquedas.size() == 0 and dirHTF != 0 and not na(n5Niv)
        if n5Niv.size() > 0
            for i = 0 to n5Niv.size() - 1
                nv5   = n5Niv.get(i)
                alto5 = n5Alto.get(i)
                clave = claveZona(2, nv5, nv5)
                // Solo manipulaciones CONFIRMADAS en 5m (cerró dentro, o volvió dentro en 4 velas).
                // Compra: manipulación de un bajo. Venta: manipulación de un alto.
                if busquedas.size() == 0 and not usadas.includes(clave) and (dirHTF == 1 ? not alto5 : alto5)
                    usadas.push(clave)
                    busquedas.push(Busqueda.new(dirHTF, 2, na, na, nv5, dirHTF == 1 ? low : high, time))
                    logE("Toca zona: " + descB(busquedas.get(0)))
    // 3b) Toques de zonas de 5m de la dirección CONTRARIA a 1H: se guardan por si 1H cambia en esta vela
    if dirHTF != 0 and not na(z5Top)
        dOp = -dirHTF
        if z5Top.size() > 0
            for i = 0 to z5Top.size() - 1
                alc = z5Alc.get(i)
                if alc == (dOp == 1)
                    t   = z5Top.get(i)
                    bt  = z5Bot.get(i)
                    fvg = z5Fvg.get(i)
                    mid = (t + bt) / 2
                    toca  = fvg ? (alc ? low <= mid : high >= mid) : (alc ? low <= t : high >= bt)
                    clave = claveZona(fvg ? 1 : 0, t, bt)
                    if toca and not usadas.includes(clave) and not estaPendiente(clave)
                        pendientes.push(Busqueda.new(dOp, fvg ? 1 : 0, t, bt, na, alc ? low : high, time))
    if dirHTF != 0 and not na(n5Niv)
        dOp = -dirHTF
        if n5Niv.size() > 0
            for i = 0 to n5Niv.size() - 1
                nv5   = n5Niv.get(i)
                alto5 = n5Alto.get(i)
                clave = claveZona(2, nv5, nv5)
                if not usadas.includes(clave) and not estaPendiente(clave) and (dOp == 1 ? not alto5 : alto5)
                    pendientes.push(Busqueda.new(dOp, 2, na, na, nv5, dOp == 1 ? low : high, time))
    while pendientes.size() > 20
        pendientes.shift()

    while usadas.size() > 200
        usadas.shift()
    if busquedas.size() == 0
        limpiarObs1()

    // Operación abierta: en cuanto el precio toca el TP o el SL se quita todo y se vuelve a buscar
    if abiertas.size() > 0
        for i = abiertas.size() - 1 to 0
            op     = abiertas.get(i)
            tocaSl = op.compra ? low <= op.sl : high >= op.sl
            tocaTp = op.compra ? high >= op.tp : low <= op.tp
            cierra = tocaSl or tocaTp
            if cierra
                logE((op.compra ? "COMPRA" : "VENTA") + (tocaSl ? " cerrada en SL" : " cerrada en TP"))
                // Historial: ✓ si tocó el TP, ✗ si tocó el SL (si toca los dos en la misma vela, cuenta como SL)
                op.marca.set_text((op.compra ? "▲" : "▼") + (tocaSl ? " ✗" : " ✓"))
                op.marca.set_color(tocaSl ? color.red : color.green)
                op.lEnt.delete()
                op.lTp.delete()
                op.lSl.delete()
                op.eEnt.delete()
                op.eTp.delete()
                op.eSl.delete()
                abiertas.remove(i)

    // 4) OB de 1 minuto: el precio vuelve a tocarlo y la primera vela que cierra más allá del cuerpo es la entrada.
    //    Solo UNA entrada por zona de 5m: después de entrar, no se busca nada más hasta tocar otra zona.
    entro = false
    if obs1.size() > 0
        for i = obs1.size() - 1 to 0
            o = obs1.get(i)
            // El stop sigue el low (compra) o high (venta) que dejó el OB, hasta la vela de entrada
            o.ext    := o.alcista ? math.min(o.ext, low) : math.max(o.ext, high)
            toca     = o.alcista ? low <= o.top : high >= o.bot
            invalida = o.alcista ? close < o.bot : close > o.top
            enZona   = low <= o.top and high >= o.bot
            o.dentro := enZona ? o.dentro + 1 : o.dentro
            entra    = (o.tocado or toca) and (o.alcista ? close > o.top : close < o.bot)
            borrar   = false
            if invalida or o.dentro > maxDentro
                logE("OB 1m invalidado " + str.tostring(o.bot, format.mintick) + "–" + str.tostring(o.top, format.mintick))
                borrar := true
            else if entra and dirHTF == (o.alcista ? 1 : -1)
                // Stop loss: justo en el low (compra) o high (venta) que dejó el OB de 1m
                sl = o.ext
                riesgo = o.alcista ? close - sl : sl - close
                if not na(sl) and riesgo > 0
                    tp = o.alcista ? close + 3 * riesgo : close - 3 * riesgo
                    lEnt = line.new(bar_index, close, bar_index + 1, close, color = color.blue, width = 2, extend = extend.right)
                    lTp  = line.new(bar_index, tp, bar_index + 1, tp, color = color.green, width = 2, extend = extend.right)
                    lSl  = line.new(bar_index, sl, bar_index + 1, sl, color = color.red, width = 2, extend = extend.right)
                    eEnt = label.new(bar_index, close, (o.alcista ? "COMPRA " : "VENTA ") + str.tostring(close, format.mintick), style = label.style_label_right, color = color.blue, textcolor = color.white, size = size.small)
                    eTp  = label.new(bar_index, tp, "TP " + str.tostring(tp, format.mintick), style = label.style_label_right, color = color.green, textcolor = color.white, size = size.small)
                    eSl  = label.new(bar_index, sl, "SL " + str.tostring(sl, format.mintick), style = label.style_label_right, color = color.red, textcolor = color.white, size = size.small)
                    // Qué zona de 5m originó la búsqueda (se ve al pasar el ratón por la marca)
                    string zonaTxt = "sin datos"
                    if busquedas.size() > 0
                        b0 = busquedas.get(0)
                        zonaTxt := b0.tipo == 2 ? ((b0.dir == 1 ? "Bajo" : "Alto") + " de 5m en " + str.tostring(b0.nivel, format.mintick)) : ((b0.tipo == 0 ? "OB" : "Vacío") + (b0.dir == 1 ? " alcista" : " bajista") + " de 5m " + str.tostring(b0.bot, format.mintick) + " – " + str.tostring(b0.top, format.mintick))
                        zonaTxt := zonaTxt + "\nTocada: " + str.format_time(b0.tInicio, "dd/MM HH:mm", syminfo.timezone)
                    tip = (o.alcista ? "COMPRA" : "VENTA")

                    abiertas.push(Operacion.new(lEnt, lTp, lSl, eEnt, eTp, eSl, tp, sl, o.alcista, marca))
                    entradaCompra := o.alcista
                    entradaVenta  := not o.alcista
                    entro := true
                    logE("ENTRADA " + (o.alcista ? "COMPRA " : "VENTA ") + str.tostring(close, format.mintick))
                borrar := true

// OBJETIVO DE LA DIRECCIÓN: el más cercano entre
//   - compra: el siguiente alto, OB bajista o vacío bajista por encima
//   - venta:  el siguiente bajo, OB alcista o vacío alcista por debajo
// Devuelve [precio donde se llega, precio de cancelación, texto]
// =====================================================================
buscarObjetivo(int d) =>
    float mejorDist = na
    float toque     = na
    float umbral    = na
    string txt      = "Sin objetivo a la vista"
    for n in niveles
        if n.vivo and (d == 1 ? (n.esAlto and n.precio > close) : (not n.esAlto and n.precio < close))
            dist = math.abs(n.precio - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                toque     := n.precio
                umbral    := n.precio
                txt       := (d == 1 ? "Alto " : "Bajo ") + str.tostring(n.precio, format.mintick)
    for z in obs
        if d == 1 ? (not z.alcista and z.bot > close) : (z.alcista and z.top < close)
            dist = math.abs((d == 1 ? z.bot : z.top) - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                // Llegar al OB = tocar su cuerpo. Pasarlo = cerrar más allá de todo el cuerpo.
                toque     := d == 1 ? z.bot : z.top
                umbral    := d == 1 ? z.top : z.bot
                txt       := (d == 1 ? "OB bajista " : "OB alcista ") + str.tostring(z.bot, format.mintick) + " – " + str.tostring(z.top, format.mintick)
    for z in fvgs
        if d == 1 ? (not z.alcista and z.bot > close) : (z.alcista and z.top < close)
            dist = math.abs((d == 1 ? z.bot : z.top) - close)
            if na(mejorDist) or dist < mejorDist
                mejorDist := dist
                // Llegar al vacío = tocarlo. Pasarlo = cerrar más allá de todo el vacío.
                toque     := d == 1 ? z.bot : z.top
                umbral    := d == 1 ? z.top : z.bot
                txt       := (d == 1 ? "Vacío bajista " : "Vacío alcista ") + str.tostring(z.bot, format.mintick) + " – " + str.tostring(z.top, format.mintick)
    [toque, umbral, txt]
