// Manipulación por estructura: barre un high/low de pivote vigente y vuelve adentro
// (misma vela, o en las siguientes 'win' velas con una vela contraria de cuerpo >= al de la ruptura).
// Si no vuelve -> quiebre real (brkUp / brkDn).
f_mani(int pl, int win, int c2w) =>
    var array<float> hs = array.new_float()
    var array<float> ls = array.new_float()
    float ph = ta.pivothigh(high, pl, pl)
    float pv = ta.pivotlow(low, pl, pl)
    if not na(ph)
        array.push(hs, ph)
        if array.size(hs) > 30
            array.shift(hs)
    if not na(pv)
        array.push(ls, pv)
        if array.size(ls) > 30
            array.shift(ls)
    float swH = na
    float swL = na
    if array.size(hs) > 0
        for i = array.size(hs) - 1 to 0
            float lv = array.get(hs, i)
            if high > lv
                swH := na(swH) ? lv : math.max(swH, lv)
                array.remove(hs, i)
    if array.size(ls) > 0
        for i = array.size(ls) - 1 to 0
            float lv = array.get(ls, i)
            if low < lv
                swL := na(swL) ? lv : math.min(swL, lv)
                array.remove(ls, i)
    float body = math.abs(close - open)
    // --- lado bajista (barre un high) ---
    var float pH = na
    var float pHx = na
    var float pHb = na
    var int pHn = 0
    bool beEvt = false
    float beX = na
    bool brkUp = false
    if not na(swH)
        if not na(pH)
            brkUp := true
        if close < swH
            beEvt := true
            beX := high
            pH := na
        else
            pH := swH
            pHx := high
            pHb := body
            pHn := 0
    else if not na(pH)
        pHn += 1
        pHx := math.max(pHx, high)
        if close < pH and close < open and body >= pHb
            beEvt := true
            beX := pHx
            pH := na
        else if pHn >= win
            brkUp := close > pH
            pH := na
    // --- lado alcista (barre un low) ---
    var float pL = na
    var float pLx = na
    var float pLb = na
    var int pLn = 0
    bool buEvt = false
    float buX = na
    bool brkDn = false
    if not na(swL)
        if not na(pL)
            brkDn := true
        if close > swL
            buEvt := true
            buX := low
            pL := na
        else
            pL := swL
            pLx := low
            pLb := body
            pLn := 0
    else if not na(pL)
        pLn += 1
        pLx := math.min(pLx, low)
        if close > pL and close > open and body >= pLb
            buEvt := true
            buX := pLx
            pL := na
        else if pLn >= win
            brkDn := close < pL
            pL := na
    // --- manipulación de la manipulación (caso 2) ---
    var float lastBuX = na
    var int lastBuI = na
    var float lastBeX = na
    var int lastBeI = na
    bool buC2 = false
    bool beC2 = false
    if buEvt
        buC2 := not na(lastBuI) and bar_index - lastBuI <= c2w and buX < lastBuX
        lastBuX := buX
        lastBuI := bar_index
    if beEvt
        beC2 := not na(lastBeI) and bar_index - lastBeI <= c2w and beX > lastBeX
        lastBeX := beX
        lastBeI := bar_index
    bool pendUp = not na(pH)
    bool pendDn = not na(pL)
    [buEvt, buX, buC2, beEvt, beX, beC2, brkUp, brkDn, pendUp, pendDn]

// Order block: vela contraria -> vela fuerte cierra más allá de su cuerpo (hasta 'within' velas después)
// -> el precio vuelve y toca el cuerpo (reacción) -> [modo entrada] primera vela que cierra más allá del cuerpo.
// Se anula si una vela cierra al otro lado del cuerpo. Un solo uso.
f_ob(bool bull, float mult, int within, bool needLiq, int maniBS, bool entryMode, int life) =>
    float body = math.abs(close - open)
    float avgB = ta.sma(body, 20)[1]
    bool strong = body >= mult * avgB
    var float cTop = na
    var float cBot = na
    var float cExt = na
    var int cIdx = na
    var int st = 0
    var float oTop = na
    var float oBot = na
    var float oExt = na
    var int oTime = na
    var int oIdx = na
    bool react = false
    bool entry = false
    bool isContra = bull ? close < open : close > open
    if st > 0
        oExt := bull ? math.min(oExt, low) : math.max(oExt, high)
        bool broken = bull ? close < oBot : close > oTop
        bool old = entryMode and bar_index - oIdx > life
        if broken or old
            st := 0
        else
            bool touch = bull ? low <= oTop : high >= oBot
            if st == 1 and touch
                st := 2
                react := true
            if st == 2 and entryMode
                bool beyond = bull ? close > oTop : close < oBot
                if beyond
                    entry := true
            if react and not entryMode
                st := 0
            if entry
                st := 0
    if not isContra and not na(cIdx) and st != 2 and not react and not entry
        bool brk = bull ? close > cTop : close < cBot
        int dist = bar_index - cIdx
        bool liqOk = not needLiq or maniBS <= dist + 3
        if brk and dist <= within and strong and liqOk
            st := 1
            oTop := cTop
            oBot := cBot
            oExt := bull ? math.min(cExt, low) : math.max(cExt, high)
            oTime := time
            oIdx := bar_index
            cIdx := na
    if isContra
        cTop := math.max(open, close)
        cBot := math.min(open, close)
        cExt := bull ? low : high
        cIdx := bar_index
    else if not na(cIdx)
        cExt := bull ? math.min(cExt, low) : math.max(cExt, high)
    [react, entry, st, oTop, oBot, oExt, oTime]

// Vacío (FVG) de 3 velas con liquidez tomada antes; da dirección cuando vuelve, toca y cierra por fuera a favor.
f_fvg(bool bull, int maniBS) =>
    var float zTop = na
    var float zBot = na
    var bool touched = false
    bool evt = false
    float inv = na
    bool formed = bull ? low > high[2] : high < low[2]
    if not na(zTop)
        bool dead = bull ? close < zBot : close > zTop
        if dead
            zTop := na
            zBot := na
        else
            bool tch = bull ? low <= zTop : high >= zBot
            if tch
                touched := true
            bool out = bull ? close > zTop : close < zBot
            if touched and out
                evt := true
                inv := bull ? zBot : zTop
                zTop := na
                zBot := na
    if formed and maniBS <= 5
        zTop := bull ? low : low[2]
        zBot := bull ? high[2] : high
        touched := false
    [evt, inv]

f_ctx1H() =>
    [mBu, mBuX, mBuC2, mBe, mBeX, mBeC2, bUp, bDn, pUp, pDn] = f_mani(piv1H, mitWin, c2Win)
    int bsBu = nz(ta.barssince(mBu), 9999)
    int bsBe = nz(ta.barssince(mBe), 9999)
    [obBuR, obBuE, obBuS, obBuT, obBuB, obBuX, obBuTm] = f_ob(true, strongHTF, obWithin, obLiq1H, bsBu, false, 0)
    [obBeR, obBeE, obBeS, obBeT, obBeB, obBeX, obBeTm] = f_ob(false, strongHTF, obWithin, obLiq1H, bsBe, false, 0)
    [fBu, fBuI] = f_fvg(true, bsBu)
    [fBe, fBeI] = f_fvg(false, bsBe)
    var int dir = 0
    var int src = 0
    var float inv = na
    if dir == 1 and (close < inv or bDn)
        dir := 0
        src := 0
    if dir == -1 and (close > inv or bUp)
        dir := 0
        src := 0
    int nd = 0
    int ns = 0
    float ni = na
    if use1HOB and obBuR
        nd := 1
        ns := 1
        ni := obBuB
    if use1HOB and obBeR
        nd := -1
        ns := 1
        ni := obBeT
    if use1HFvg and fBu
        nd := 1
        ns := 3
        ni := fBuI
    if use1HFvg and fBe
        nd := -1
        ns := 3
        ni := fBeI
    if use1HMani and mBu and not mBe
        nd := 1
        ns := 2
        ni := mBuX
    if use1HMani and mBe and not mBu
        nd := -1
        ns := 2
        ni := mBeX
    if nd != 0
        dir := nd
        src := ns
        inv := ni
    bool pend = (dir == 1 and pDn) or (dir == -1 and pUp)
    [dir[1], src[1], inv[1], pend[1], time_close[1]]

f_minAbove(array<float> a, float lv) =>
    float r = na
    for v in a
        if v > lv and (na(r) or v < r)
            r := v
    r

f_maxBelow(array<float> a, float lv) =>
    float r = na
    for v in a
        if v < lv and (na(r) or v > r)
            r := v
    r

f_sess() =>
    int d = dayofweek(time, TZ)
    int mins = hour(time, TZ) * 60 + minute(time, TZ)
    bool wk = d >= dayofweek.monday and d <= dayofweek.friday
    bool ny = useNY and wk and mins >= 390 and mins < 720
    bool asiaDay = d == dayofweek.sunday or (d >= dayofweek.monday and d <= dayofweek.thursday)
    bool asia = useAsia and asiaDay and mins >= asiaFrom * 60 and mins < asiaTo * 60
    ny or asia

    t
